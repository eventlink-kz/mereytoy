RSVP — подключение общей базы

Сайт уже работает локально. Чтобы ответы со всех телефонов попадали в одну админку, подключите бесплатную Google Таблицу через Apps Script.

1. Google Таблица

  1. Создайте пустую Google Таблицу.
  2. Откройте Расширения → Apps Script.
  3. Вставьте код:
const SHEET_NAME = 'RSVP';
const ADMIN_KEY = '8080';

function doGet(e) {
  if (String(e.parameter.action || '') !== 'list') return json({ok:false,error:'Unknown action'});
  if (String(e.parameter.key || '') !== ADMIN_KEY) return json({ok:false,error:'Unauthorized'});
  const sh = getSheet();
  const values = sh.getDataRange().getValues();
  const entries = values.slice(1).filter(r => r[0]).map(r => ({
    id:r[0], createdAt:r[1], name:r[2], attendance:r[3],
    guests:r[4], phone:r[5], comment:r[6]
  })).reverse();
  return json({ok:true, entries});
}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents || '{}');
    const x = body.entry || {};
    if (body.action !== 'submit' || !x.name || !['yes','no'].includes(x.attendance)) {
      return json({ok:false,error:'Validation error'});
    }
    getSheet().appendRow([
      String(x.id || Utilities.getUuid()),
      String(x.createdAt || new Date().toISOString()),
      String(x.name).slice(0,80),
      x.attendance,
      x.attendance === 'yes' ? Number(x.guests || 1) : 0,
      String(x.phone || '').slice(0,30),
      String(x.comment || '').slice(0,300)
    ]);
    return json({ok:true});
  } catch (err) {
    return json({ok:false,error:String(err)});
  }
}

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) sh = ss.insertSheet(SHEET_NAME);
  if (sh.getLastRow() === 0) sh.appendRow(['id','createdAt','name','attendance','guests','phone','comment']);
  return sh;
}
function json(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

2. Публикация

  1. Deploy → New deployment → Web app.
  2. Execute as: Me.
  3. Who has access: Anyone.
  4. Скопируйте URL, заканчивающийся на /exec.

3. Подключение

В этой версии сайт уже подключён к Google Apps Script.

Важно: в настройках развертывания Apps Script доступ должен быть установлен как Anyone / Все.

PIN админки по умолчанию: 8080. Для семейного сайта этого достаточно как лёгкого ограничения доступа, но это не полноценная серверная авторизация.
